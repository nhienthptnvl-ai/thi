  <script>

    // ========== THÔNG TIN CỦA THẦY ==========

    const SUPABASE_URL = 'https://wdjkafeuiufgtbklada.supabase.co';

    const SUPABASE_KEY = 'sb_publishable_JjUILVUtoUY-_DcZVYRneg_87UpV6Z2';

    // ========================================

 

    let supabaseClient = null;

    let deChon = null;

 

    // === TẢI THƯ VIỆN TRỰC TIẾP KHÔNG CẦN LIÊN KẾT ===

    async function taiThuVien() {

      const el = document.getElementById('trangthai');

      el.innerHTML = '🔄 Đang tải thư viện...';

     

      try {

        const res = await fetch('https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2/dist/supabase.min.js');

        if (!res.ok) throw new Error('Không tải được');

        const code = await res.text();

        const script = document.createElement('script');

        script.textContent = code;

        document.head.appendChild(script);

        return true;

      } catch {

        try {

          const res = await fetch('https://unpkg.com/@supabase/supabase-js@2.45.1/dist/umd/supabase.min.js');

          if (!res.ok) throw new Error('Không tải được');

          const code = await res.text();

          const script = document.createElement('script');

          script.textContent = code;

          document.head.appendChild(script);

          return true;

        } catch (e) {

          el.className = 'trangthai sai';

          el.innerHTML = `❌ Không tải được thư viện<br>

            <div class="huongdan">

            💡 HTMLPut chặn tải từ ngoài → Dùng Netlify thay thế:<br>

            ✅ Vào app.netlify.com → Đăng nhập → Kéo thả file lên → Mở link đó

            </div>`;

          return false;

        }

      }

    }

 

    async function khoiTao() {

      const el = document.getElementById('trangthai');

     

      if (!await taiThuVien()) return;

     

      if (typeof supabase === 'undefined') {

        el.className = 'trangthai sai';

        el.innerHTML = '❌ Thư viện chưa sẵn sàng → Tải lại trang';

        return;

      }

     

      el.innerHTML = '🔄 Đang kết nối CSDL...';

     

      try {

        supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_KEY, {

          auth: { persistSession: false }

        });

 

        const { data, error } = await supabaseClient

          .from('de_kiem_tra')

          .select('ma_de, ten_de')

          .limit(100);

 

        if (error) throw error;

 

        el.className = 'trangthai tot';

        el.innerHTML = `✅ Kết nối thành công — ${data.length} đề`;

        document.getElementById('btn-giaovien').classList.remove('an');

 

        const ma = new URLSearchParams(window.location.search).get('m');

        if (ma) {

          document.getElementById('ma_de').value = ma.toUpperCase();

          kiemTraMa();

        }

 

      } catch (err) {

        el.className = 'trangthai sai';

        el.innerHTML = `❌ Lỗi: ${err.message}<br>

          <div class="huongdan">

          1. Đã chạy SQL tạo 3 bảng trên Supabase chưa?<br>

          2. URL & KEY đúng chưa?

          </div>`;

      }

    }

 

    async function kiemTraMa() {

      const ma = document.getElementById('ma_de').value.trim();

      const tb = document.getElementById('thongbao');

      const btn = document.getElementById('vao-lambai');

      tb.innerHTML = ''; btn.classList.add('an'); deChon = null;

 

      if (!ma) return tb.innerHTML = '<p style="color:#e74c3c">⚠️ Nhập mã đề</p>';

      if (!supabaseClient) return tb.innerHTML = '<p style="color:#e74c3c">❌ Chưa kết nối</p>';

 

      const { data, error } = await supabaseClient

        .from('de_kiem_tra')

        .select('*')

        .eq('ma_de', ma)

        .single();

 

      if (error || !data) {

        tb.innerHTML = `<p style="color:#e74c3c">❌ Mã đề ${ma} không tồn tại</p>`;

        return;

      }

 

      deChon = data;

      tb.innerHTML = `<p style="color:#27ae60">✅ ${data.ten_de} — ${data.cau_hoi.length} câu</p>`;

      btn.classList.remove('an');

    }

 

    function vaoLamBai() {

      if (!deChon) return alert('Kiểm tra mã đề trước!');

      alert('✅ Đã vào làm bài — phát triển tiếp');

    }

 

    function vaoGiaoVien() {

      alert('✅ Giao diện Giáo viên — Tạo đề & Quản lý điểm');

    }

 

    khoiTao();
